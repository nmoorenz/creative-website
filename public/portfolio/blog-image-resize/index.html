height_width <- function(filenames) {
  h = array(dim = length(filenames))
  w = array(dim = length(filenames))
  for (i in seq_along(filenames)) {
    k = filenames[i]
    if (str_detect(k, '.png') || str_detect(k, '.jpg')) {
      img = read.bitmap(k)
      h[i] = dim(img)[1]
      w[i] = dim(img)[2]
    } else {
      h[i] = 0
      w[i] = 0
    }
  }
  list(height = h, width = w)
}

from PIL import Image

def height_width(filename): 
    # filename = row['filename']
    if '.jpg' in filename or '.png' in filename:
        with Image.open(filename) as img: 
            width, height = img.size
    else: 
        width, height = 0, 0
    return width, height
